## TICKETS

```dsl
RULE C-TICKET-001
WHEN TASK_STEPS > 1 OR AGENT_REQUIRED
DO ASSIGN UNIQUE TICKET_ID
DO CREATE_OR_UPDATE TICKET_DIRECTORY_PATTERN
DO CREATE_OR_UPDATE TICKET_MAIN_FILE
ASSERT TASK_SCOPE_RECORDED_IN_SINGLE_TICKET_DIRECTORY

RULE C-TICKET-002
WHEN PARTICIPANT_ASSIGNED
DO CREATE PARTICIPANT_FILE MATCHING TICKET_PARTICIPANT_PATTERN
ASSERT ONLY_ASSIGNED_PARTICIPANTS_HAVE_FILES

RULE C-TICKET-003
WHEN TICKET_CREATED
DO RECORD GOAL SCOPE RISKS ACCEPTANCE_CRITERIA STATUS IN TICKET_MAIN_FILE
ASSERT TICKET_MAIN_FILE_IS_SCOPE_SOURCE_OF_TRUTH

RULE C-TICKET-004
WHEN PARTICIPANT_WORKS_ON_TICKET
DO RECORD INSTRUCTIONS PLAN ACTUAL_CHANGES UNFINISHED_ITEMS BLOCKERS IN PARTICIPANT_FILE
ASSERT PARTICIPANT_FILE_DOES_NOT_REDEFINE_TICKET_SCOPE

RULE C-TICKET-005
WHEN COMMAND_TEST_OR_RUNTIME_OUTPUT_PRODUCED
DO APPEND RAW_OUTPUT_TO TICKET_LOG_FILE
ASSERT RAW_LOGS_PRESERVED_IN_TEXT_FORMAT

RULE C-TICKET-006
WHEN PROJECT_TODO_REFERENCES_TICKET
DO REFERENCE TICKET_DIRECTORY_PATTERN
ASSERT TODO_CONTAINS_PROJECT_LEVEL_INDEX_ONLY

RULE C-TICKET-007
WHEN TICKET_WORK_COMPLETED
DO PRESERVE_TICKET_DIRECTORY
ASSERT TICKET_DELETION_FORBIDDEN_WITHOUT_EXPLICIT_POST_PROJECT_USER_REQUEST
```

## DOCKER

```dsl
DOCKER_REQUIRED_FILES = [
  "Dockerfile",
  "compose.yml OR docker-compose.yml"
]

RULE C-DOCKER-001
WHEN TASK_RECEIVED
DO VERIFY "docker" COMMAND_EXISTS
DO VERIFY DOCKER_ENGINE_RUNNING
ASSERT DOCKER_STATUS = VERIFIED
NEXT ANALYSIS OR BLOCKED

RULE C-DOCKER-002
WHEN PROJECT_EXECUTION_REQUIRED
DO VERIFY DOCKER_REQUIRED_FILES
ASSERT PROJECT_RUNTIME_DEFINED_IN_CONTAINERS
NEXT PLAN OR BLOCKED

RULE C-DOCKER-003
WHEN REQUIRED_RUNTIME_TOOL_OR_SERVICE_USED
DO RUN_THROUGH_PROJECT_DOCKER_ENVIRONMENT
ASSERT HOST_TOOL_VERSION_NOT_USED_AS_PROJECT_SOURCE_OF_TRUTH
NEXT VALIDATION

RULE C-DOCKER-004
WHEN DOCKER_IMAGE_BUILT
DO USE_PINNED_BASE_IMAGE_VERSION
ASSERT RUNTIME_VERSION_REPRODUCIBLE
NEXT VALIDATION

RULE C-DOCKER-005
WHEN DEPENDENCY_OR_RUNTIME_VERSION_CHANGED
DO UPDATE DOCKER_CONFIGURATION
DO RECORD CHANGE_IN_TICKET_PARTICIPANT_FILE
ASSERT CONTAINER_ENVIRONMENT_MATCHES_PROJECT_REQUIREMENTS
NEXT VALIDATION

RULE C-DOCKER-006
WHEN DOCKER_NOT_INSTALLED_OR_ENGINE_NOT_RUNNING
DO SET RESULT = BLOCKED
DO REPORT INSTALLATION_OR_STARTUP_REQUIREMENT
ASSERT LOCAL_FALLBACK_NOT_USED
NEXT BLOCKED
```

## STATE

```dsl
STATE START
STATE ANALYSIS
STATE PLAN
STATE TOOLS
STATE DELEGATION
STATE EDIT
STATE VALIDATION
STATE PUBLICATION
STATE DONE
STATE BLOCKED

TRANSITION START -> ANALYSIS
TRANSITION ANALYSIS -> PLAN
TRANSITION PLAN -> TOOLS
TRANSITION TOOLS -> DELEGATION
TRANSITION DELEGATION -> EDIT
TRANSITION EDIT -> VALIDATION
TRANSITION VALIDATION -> PUBLICATION
TRANSITION PUBLICATION -> DONE
TRANSITION ANY -> BLOCKED WHEN REQUIRED_EVIDENCE_MISSING
```

## PROCEDURA START

```dsl
RULE C-START-001
WHEN TASK_RECEIVED
DO READ_RELEVANT_FILES
DO RUN "git status --short"
DO VERIFY "docker" COMMAND_EXISTS
DO VERIFY DOCKER_ENGINE_RUNNING
ASSERT WORKTREE_STATE_CAPTURED
ASSERT DOCKER_STATUS = VERIFIED
NEXT ANALYSIS

RULE C-START-002
WHEN TASK_SCOPE_NOT_CLEAR
DO LIMIT_SCOPE_TO_USER_REQUEST
ASSERT SCOPE_MINIMAL
NEXT ANALYSIS

RULE C-START-003
WHEN TASK_STEPS > 1 OR AGENT_REQUIRED
DO IDENTIFY TICKET_ID OWNER AND PARTICIPANTS
DO READ TICKET_MAIN_FILE IF EXISTS
DO READ RELEVANT_PARTICIPANT_FILES IF EXISTS
ASSERT TICKET_CONTEXT_CAPTURED
NEXT ANALYSIS
```

## PROCEDURA ANALYSIS

```dsl
RULE C-ANALYSIS-001
WHEN CLAIM_OR_TOOL_OR_WORKFLOW_REFERENCED
DO VERIFY_IN_CURRENT_FILES_OR_COMMAND_OUTPUT
ASSERT UNVERIFIED_ITEMS_ARE_LABELLED
NEXT PLAN

RULE C-ANALYSIS-002
WHEN project.sh IS_RELEVANT
DO DISTINGUISH ACTIVE_COMMANDS FROM COMMENTED_COMMANDS
ASSERT COMMENTED_COMMANDS_NOT_USED
NEXT PLAN

RULE C-ANALYSIS-003
WHEN REPOSITORY_FEATURE_NOT_FOUND
DO MARK FEATURE AS UNVERIFIED
ASSERT FEATURE_NOT_PRESENTED_AS_AVAILABLE
NEXT PLAN

RULE C-ANALYSIS-004
WHEN PROJECT_EXECUTION_OR_TESTS_REQUIRED
DO VERIFY DOCKER_CONFIGURATION
DO VERIFY REQUIRED_SERVICES_DEFINED
ASSERT EXECUTION_ENVIRONMENT_REPRODUCIBLE
NEXT PLAN OR BLOCKED
```

## PROCEDURA PLAN

```dsl
RULE C-PLAN-001
WHEN TASK_STEPS <= 1 AND AGENT_NOT_REQUIRED
DO KEEP_PLAN_INLINE
ASSERT TODO_NOT_REQUIRED
NEXT TOOLS

RULE C-PLAN-002
WHEN TASK_STEPS > 1 OR AGENT_REQUIRED
DO CREATE_OR_UPDATE "TODO.md"
DO CREATE_OR_UPDATE TICKET_DIRECTORY_PATTERN
DO CREATE_OR_UPDATE TICKET_MAIN_FILE
DO RECORD SCOPE RISKS ACCEPTANCE_CRITERIA STATUS
ASSERT TODO_CONTAINS_PROJECT_LEVEL_WORK_TRACKING
ASSERT TICKET_MAIN_FILE_CONTAINS_ASSIGNED_SCOPE
NEXT TOOLS

RULE C-PLAN-003
WHEN CHANGE_RISK_HIGH
DO RECORD RISK_AND_CHECKS
ASSERT RISK_VISIBLE_BEFORE_EDIT
NEXT TOOLS

RULE C-PLAN-004
WHEN PARTICIPANT_ASSIGNED
DO CREATE_OR_UPDATE PARTICIPANT_FILE
DO RECORD ROLE INSTRUCTIONS AND IMPLEMENTATION_PLAN
ASSERT IMPLEMENTATION_PLAN_DOES_NOT_REDEFINE_TICKET_SCOPE
NEXT TOOLS
```

## PROCEDURA TOOLS

```dsl
RULE C-TOOLS-001
WHEN TOOL_REQUIRED
DO VERIFY TOOL_EXISTS AND INVOCATION_EXISTS
ASSERT TOOL_STATUS IN [VERIFIED, UNVERIFIED]
NEXT DELEGATION

RULE C-TOOLS-002
WHEN PACKAGE_INSTALLED_AND_COMMAND_NOT_INVOKED
DO SET TOOL_STATUS = INSTALL_ONLY
ASSERT TOOL_NOT_PRESENTED_AS_AVAILABLE
NEXT DELEGATION

RULE C-TOOLS-003
WHEN DOCUMENTATION_ONLY_CHANGE
DO SKIP "project.sh"
ASSERT INSTALL_OR_OVERWRITE_NOT_REQUIRED
NEXT DELEGATION

RULE C-TOOLS-004
WHEN PROJECT_COMMAND_REQUIRED
DO USE DOCKER_COMPOSE_OR_DOCUMENTED_DOCKER_COMMAND
ASSERT COMMAND_EXECUTED_IN_PROJECT_CONTAINER
NEXT DELEGATION OR BLOCKED

RULE C-TOOLS-005
WHEN AGENT_WORKS_ON_ANALYSIS_OR_REFACTOR
DO RUN_DEV_TOOLS_AUTOMATION ("./project.sh" OR "project.bat")
DO READ_GENERATED_REPORT_FROM_PROJECT_DIR
ASSERT DEV_TOOLS_AUTOMATION_PRIORITIZED_OVER_MANUAL_TOKEN_BURNING
NEXT DELEGATION
```

## PROCEDURA DELEGATION

```dsl
RULE C-DELEGATION-001
WHEN AGENT_REQUIRED
DO VERIFY AGENT_DEFINITION AND AGENT_INVOCATION
ASSERT AGENT_STATUS IN [VERIFIED, UNVERIFIED]
NEXT EDIT

RULE C-DELEGATION-002
WHEN AGENT_STATUS = UNVERIFIED
DO USE_LOCAL_WORKFLOW_OR_REPORT_BLOCKER
ASSERT UNVERIFIED_AGENT_NOT_USED
NEXT EDIT OR BLOCKED

RULE C-DELEGATION-003
WHEN AGENT_REQUIRED
DO REQUIRE TICKET_MAIN_FILE
DO REQUIRE ASSIGNED_PARTICIPANT_FILE
DO RECORD MATERIAL_AGENT_INSTRUCTIONS
ASSERT AGENT_INSTRUCTIONS_REFERENCE_TICKET
NEXT EDIT

RULE C-DELEGATION-004
WHEN DELEGATING_TO_SUBAGENT
DO PASS_MINIMAL_TRIMMED_CONTEXT
DO VERIFY_SUBAGENT_OUTPUT_IN_PARENT
ASSERT SUBAGENT_RESULT_VALIDATED_BEFORE_MERGE
NEXT EDIT
```

## PROCEDURA EDIT

```dsl
RULE C-EDIT-001
WHEN TARGET_FILE_EXISTS
DO READ TARGET_FILE
DO MODIFY_ONLY_REQUESTED_SCOPE
ASSERT EXISTING_MEANING_PRESERVED
NEXT VALIDATION

RULE C-EDIT-002
WHEN TARGET_FILE_MISSING
DO VERIFY_PARENT_DIRECTORY
DO CREATE_ONLY_IF_REQUIRED_BY_TASK
ASSERT NEW_FILE_JUSTIFIED
NEXT VALIDATION

RULE C-EDIT-003
WHEN DOCUMENTATION_CHANGED
DO KEEP_TERMS_CONSISTENT
DO REMOVE_DUPLICATION
ASSERT DOCUMENTATION_SHORTER_OR_CLEARER
NEXT VALIDATION

RULE C-EDIT-004
WHEN TICKET_USED
DO UPDATE CURRENT_PARTICIPANT_FILE
DO RECORD ACTUAL_CHANGES
DO RECORD UNFINISHED_ITEMS
DO RECORD BLOCKERS
ASSERT EXECUTION_STATE_VISIBLE
NEXT VALIDATION

RULE C-EDIT-005
WHEN COMMAND_OUTPUT_OR_TEST_OUTPUT_EXISTS
DO APPEND RAW_OUTPUT_TO TICKET_LOG_FILE
ASSERT LOG_OUTPUT_NOT_REPLACED_BY_SUMMARY
NEXT VALIDATION
```

## PROCEDURA VALIDATION

```dsl
RULE C-VALIDATION-001
WHEN CHANGE_COMPLETE
DO REVIEW_DIFF
DO CHECK_FORMAT
DO CHECK_REFERENCES
DO CHECK_SECRETS
ASSERT CHANGE_REVIEWED
NEXT PUBLICATION

RULE C-VALIDATION-002
WHEN TESTS_EXIST_AND_RELEVANT
DO RUN_RELEVANT_TESTS_IN_DOCKER
DO APPEND RAW_TEST_OUTPUT_TO TICKET_LOG_FILE
ASSERT TEST_RESULT_REPORTED
ASSERT HOST_RUNTIME_NOT_USED_FOR_PROJECT_VALIDATION
NEXT PUBLICATION

RULE C-VALIDATION-003
WHEN TESTS_NOT_FOUND_OR_NOT_RELEVANT
DO REPORT TESTS_NOT_RUN_WITH_REASON
ASSERT TEST_SUCCESS_NOT_CLAIMED
NEXT PUBLICATION

RULE C-VALIDATION-004
WHEN TICKET_USED
DO COMPARE TICKET_MAIN_FILE
DO COMPARE PARTICIPANT_FILES
DO COMPARE CURRENT_DIFF
DO COMPARE TEST_RESULTS
DO RECORD ACCEPTANCE_CRITERIA_EVIDENCE
ASSERT COMPLETED_AND_UNFINISHED_SCOPE_VISIBLE
NEXT PUBLICATION

RULE C-VALIDATION-005
WHEN DOCKER_CONFIGURATION_EXISTS
DO VALIDATE CONTAINER_BUILD
DO VALIDATE REQUIRED_SERVICES_START
ASSERT DOCKER_ENVIRONMENT_OPERATIONAL
NEXT PUBLICATION
```

## PROCEDURA PUBLICATION

```dsl
RULE C-PUBLISH-001
WHEN USER_DID_NOT_REQUEST_COMMIT
DO SKIP COMMIT
DO SKIP PUSH
ASSERT GIT_HISTORY_UNCHANGED
NEXT DONE

RULE C-PUBLISH-002
WHEN USER_REQUESTS_COMMIT
DO REVIEW "git status --short"
DO REVIEW "git diff"
DO COMMIT_ATOMIC_CHANGE
ASSERT COMMIT_DESCRIBES_REASON
NEXT DONE

RULE C-PUBLISH-003
WHEN USER_REQUESTS_PUSH
DO PUSH_ONLY_AFTER_COMMIT
ASSERT PUSH_TARGET_EXPLICIT
NEXT DONE
```

## FUNKCJA DONE

```dsl
DONE WHEN
  USER_SCOPE_COMPLETED
  AND TICKET_SCOPE_VALIDATED_WHEN_REQUIRED
  AND DIFF_REVIEWED
  AND SECRET_CHECK_DONE
  AND CLAIMS_VERIFIED_OR_LABELLED
  AND DOCKER_STATUS_VERIFIED
  AND DOCKER_VALIDATION_REPORTED_WHEN_RELEVANT
  AND TEST_STATUS_REPORTED
  AND COMMIT_PUSH_STATUS_REPORTED

REPORT MUST_INCLUDE [
  SUMMARY,
  TICKET_REFERENCE,
  PARTICIPANTS,
  COMPLETED_SCOPE,
  UNFINISHED_SCOPE,
  CHANGED_BLOCKS,
  PRESERVED_ELEMENTS,
  CHECKS_RUN,
  DOCKER_STATUS,
  TEST_STATUS,
  LOG_FILE_REFERENCE,
  LIMITATIONS
]
```
